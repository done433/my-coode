sudo apt install sysstat -y        # התקנת חבילת sysstat לניטור ביצועים (CPU, דיסק, רשת)

nano /etc/default/sysstat          # עריכת קובץ הגדרות להפעלת איסוף הנתונים
ENABLED="true"                    # הפעלת איסוף נתונים אוטומטי

sudo systemctl enable sysstat      # הפעלת השירות בעלייה אוטומטית של המערכת
sudo systemctl start sysstat       # התחלת השירות בפועל

systemctl list-timers | grep sysstat   # בדיקה שהטיימרים (cron) של sysstat פעילים

cat /etc/cron.d/sysstat           # צפייה במשימות המתוזמנות של sysstat

ls -l /var/log/sa/                # צפייה בקבצי הלוג שנשמרים ע"י sysstat

iostat                            # סטטיסטיקות דיסק ו-I/O
