#מתקין Docker
sudo dnf install docker-ce docker-ce-cli containerd.io -y

#מפעיל ומגדיר לעלות אוטומטית
sudo systemctl enable --now docker

#מוסיף את המשתמש שלך כדי לא לכתוב sudo כל פעם

#=== siege — בדיקת עומסים על כמה דפים במקביל ===
apt install siege -y

#מריץ על רשימת URLs
siege -c 50 -t 30S -f urls.txt





#=== בדיקת עומסים על כמה דפים ===
#כותב קובץ עם רשימת URLs
cat > urls.txt << EOF
http://localhost/
http://localhost/data.json
http://localhost/map.html
http://localhost/uploads/
EOF

#מריץ את הבדיקה על כל URL בתורו
while read url; do
    echo "=== בודק $url ==="
    ab -n 1000 -c 50 "$url"
done < urls.txt




#== התקנה ושימוש ב-ab (Apache Benchmark) ===
