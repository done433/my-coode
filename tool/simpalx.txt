import sqlite3, binascii, json

db_path = "/home/ubuntu/test/.simplex/simplex_v1_chat.db"
conn = sqlite3.connect(db_path)
cur = conn.cursor()

messages = []
cur.execute("SELECT message_id, hex(msg_body), created_at FROM messages;")
for msg_id, hex_blob, created_at in cur.fetchall():
    try:
        text = bytes.fromhex(hex_blob).decode('utf-8')
        data = json.loads(text)
        messages.append({
            "id": msg_id,
            "created_at": created_at,
            "message": data
        })
    except Exception as e:
        messages.append({
            "id": msg_id,
            "created_at": created_at,
            "error": str(e)
        })

print(json.dumps(messages, indent=2))
