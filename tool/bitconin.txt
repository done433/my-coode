openssl ecparam -name secp256k1 -genkey -noout -out ~/mycoin-key.pem

PUBKEY=$(openssl ec -in ~/mycoin-key.pem -pubout -conv_form compressed -outform DER 2>/dev/null | tail -c 33 | od -An -tx1 | tr -d ' \n')

read GTIME GNONCE GHASH GMERKLE < <(python3 - "$MSG" "$PUBKEY" "$REWARD" <<'PYEOF'
import hashlib, struct, sys, time
msg, pub, reward = sys.argv[1].encode(), bytes.fromhex(sys.argv[2]), int(sys.argv[3])
d = lambda b: hashlib.sha256(hashlib.sha256(b).digest()).digest()
def push(x):
    return (bytes([len(x)]) if len(x) < 76 else b"\x4c" + bytes([len(x)])) + x
sig = push(struct.pack("<I", 486604799)) + push(b"\x04") + push(msg)
spk = push(pub) + b"\xac"
tx = (struct.pack("<i", 1) + b"\x01" + b"\x00"*32 + b"\xff"*4 + bytes([len(sig)]) + sig + b"\xff"*4
      + b"\x01" + struct.pack("<q", reward * 10**8) + bytes([len(spk)]) + spk + b"\x00"*4)
merkle = d(tx)
bits = 0x1e0fffff
target = (bits & 0xffffff) << (8 * ((bits >> 24) - 3))
t, n = int(time.time()), 0
while True:
    h = d(struct.pack("<i", 1) + b"\x00"*32 + merkle + struct.pack("<III", t, bits, n))
    if int.from_bytes(h, "little") <= target:
        break
    n += 1
print(t, n, h[::-1].hex(), merkle[::-1].hex())
PYEOF
)
echo "Time=$GTIME Nonce=$GNONCE"
echo "Hash=$GHASH"
echo "Merkle=$GMERKLE"
