#!/usr/bin/env python3

import sys
import subprocess
import re
import sympy as s

cmd = ["operon_nsgp"] + sys.argv[1:]

p = subprocess.Popen(
    cmd,
    stdout=subprocess.PIPE,
    stderr=subprocess.STDOUT,
    text=True
)

lines = []

for line in p.stdout:
    lines.append(line)

p.wait()

# בדיוק כמו grep '^(' | tail -1
expressions = [
    line.strip()
    for line in lines
    if line.strip().startswith("(") and line.strip().endswith(")")
]

if not expressions:
    print("לא נמצאה נוסחה")
    sys.exit(1)

expr = expressions[-1]

try:
    x = s.symbols("x")
    result = s.sympify(expr)
    result = s.expand(result)
    result = s.simplify(result)

    print("y =", result)

except Exception as e:
    print("שגיאה בפענוח הנוסחה:")
    print(expr)
    print(e)
