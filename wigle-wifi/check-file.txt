/usr/bin/python3 -c "
import xml.etree.ElementTree as ET, json

tree = ET.parse('fixed2.kml')
ns = {'kml': 'http://www.opengis.net/kml/2.2'}
features = []
for pm in tree.findall('.//kml:Placemark', ns):
    name = pm.findtext('kml:name', '', ns)
    desc = pm.findtext('kml:description', '', ns) or ''
    coords = (pm.findtext('.//kml:coordinates', '', ns) or '').strip().split(',')
    props = {'SSID': name}
    for line in desc.strip().splitlines():
        if ':' in line:
            k,v = line.split(':',1)
            props[k.strip()] = v.strip()
    if len(coords) >= 2:
        try:
            features.append({'type':'Feature','geometry':{'type':'Point','coordinates':[float(coords[0]),float(coords[1])]},'properties':props})
        except: pass

print(json.dumps({'type':'FeatureCollection','features':features},ensure_ascii=False,indent=2))
" > output.geojson
