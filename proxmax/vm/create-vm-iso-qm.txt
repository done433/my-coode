#list vm
qm list

# 1. יצירת VM
qm create 100 --name debian13 --memory 2048 --cores 2 --net0 virtio,bridge=vmbr0

# 2. ייבוא דיסק ה-QCOW2
qm importdisk 100 /var/lib/vz/template/iso/debian-13-generic-amd64.qcow2 local

# 3. חיבור הדיסק ל-VM
qm set 100 --scsihw virtio-scsi-pci --scsi0 local:100/vm-100-disk-0.raw

# 4. הגדרת הדיסק כדיסק אתחול
qm set 100 --boot order=scsi0

# 5. הוספת Cloud-Init
qm set 100 --ide2 local:cloudinit

# 6. יצירת משתמש
qm set 100 --ciuser debian

# 7. הגדרת סיסמה
# החלף את YourStrongPassword123! בסיסמה שלך
qm set 100 --cipassword 

# 8. הוספת Serial Console
qm set 100 --serial0 socket

# 9. בדיקת ההגדרות
qm config 100

# 10. הפעלת ה-VM
qm start 100

# 11. התחברות ל-Console
qm terminal 100
