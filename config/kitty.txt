# --- העתק / הדבק ---
# Ctrl+C:
#   אם יש טקסט מסומן -> העתק
#   אם אין -> Ctrl+C אמיתי (SIGINT לעצור תוכנית רצה)
# זה המיפוי הפשוט והבטוח שביקשת:
map ctrl+c copy_or_interrupt

# דוגמה חלופית מתקדמת (כבויה) – קודם להעתיק, אחרת לשלוח Ctrl+C ידני:
# map ctrl+c combine : copy_to_clipboard : send_text normal,application \x03

# Ctrl+V מדביק מהקליפבורד:
map ctrl+v paste_from_clipboard

# לא צריך Ctrl+Shift+C/V:
unmap ctrl+shift+c
unmap ctrl+shift+v

# סימנת עם העכבר? זה מועתק אוטומטית:
copy_on_select yes


# --- ניווט טקסט (תזוזה כמו מק / יוניקס מהיר) ---
# קפיצה מילה אחורה/קדימה (Option+Arrow במק -> Alt+Arrow כאן):
map alt+left  send_text all \x1bb
map alt+right send_text all \x1bf

# מחיקת מילה אחורה (Option+Backspace במק):
map alt+backspace send_text all \x1b\x7f

# מחיקת מילה קדימה (Option+D במק):
map alt+d      send_text all \x1bd
map alt+delete send_text all \x1bd

# תחילת/סוף שורה:
map ctrl+a send_text all \x01
map ctrl+e send_text all \x05

# מחוק עד ההתחלה / עד הסוף:
map ctrl+u send_text all \x15
map ctrl+k send_text all \x0b

# מחיקת מילה אחורה (קלאסי יוניקס):
map ctrl+w send_text all \x17

# חיפוש בתוך הפלט על המסך (כמו Ctrl+F באפליקציות):
map ctrl+f        start_search
map ctrl+shift+f  search_next
map ctrl+shift+r  search_prev


# --- בחירה עם מקלדת בלבד (ללא עכבר) ---
# Alt+S → התחל בחירה + העתקה:
map alt+s select_and_copy_to_clipboard


# --- גלילה בפלט (במקום PageUp/PageDown מכוער) ---
map ctrl+up           scroll_line_up
map ctrl+down         scroll_line_down
map ctrl+shift+up     scroll_page_up
map ctrl+shift+down   scroll_page_down


# --- טאבים / חלונות / ניווט ---
# טאבים כמו דפדפן:
map ctrl+t        new_tab
map ctrl+w        close_tab
map ctrl+shift+]  next_tab
map ctrl+shift+[  previous_tab

# חלון חדש / סגירת חלון (טרמינל חדש / סגור):
map ctrl+n        new_window
map ctrl+q        close_window

# פיצולים פנימיים (split) ולזוז ביניהם עם Ctrl+Alt+חצים:

map ctrl+alt+left  neighboring_window left
map ctrl+alt+right neighboring_window right
map ctrl+alt+up    neighboring_window up
map ctrl+alt+down  neighboring_window down


# --- גודל טקסט על המסך (כמו דפדפן) ---
